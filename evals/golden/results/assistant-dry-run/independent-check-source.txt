import ast
import re
import datetime as dt
import hashlib
import json
import os
import pathlib
import shutil
import subprocess
import sys
import tempfile

ROOT=pathlib.Path('/Users/ceewai/Documents/SMU/Y4S1/CS467/Assignment3/repo')
BASE=ROOT/'evals/golden/results/assistant-dry-run'
MAN=json.loads((BASE/'manifest.json').read_text())
SEAL=json.loads((BASE/'sealed-proposals.json').read_text())
TEMP=pathlib.Path('/var/folders/bg/2dlhd5nj7wlbvx44ry7lmbbh0000gn/T/cs467-assistant-dry-run-as9ki3dr')
SOURCE=MAN['source_commit']
sha=lambda b:hashlib.sha256(b).hexdigest()
def git_bytes(path,rev=SOURCE):
 return subprocess.check_output(['git','show',f'{rev}:{path}'],cwd=ROOT)
def check(condition,message):
 if not condition: raise AssertionError(message)

RUNTIME={p:git_bytes(p) for p in MAN['runtime_sha256']}
for p,b in RUNTIME.items():check(sha(b)==MAN['runtime_sha256'][p],f'runtime hash: {p}')
for p in ['evals/tools/score.py','evals/golden/RUBRIC.md']:
 check(git_bytes(p)==git_bytes(p,'4baa2cb'),f'frozen changed {p}')
check((TEMP/'recorder.py').read_bytes()==(ROOT/'evals/tools/offline_dry_run.py').read_bytes(),'recorder export differs')
check((TEMP/'PROTOCOL.md').read_bytes()==(BASE/'protocol.md').read_bytes(),'protocol export differs')
tree=ast.parse(git_bytes('evals/tools/replay.py').decode())
neutralise_node=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='neutralise')
exec(compile(ast.Module(body=[neutralise_node],type_ignores=[]),'neutralise-from-source','exec'))
ALL_EVENTS=[]
SUMMARY=[]
BATCH_END={}
script_count=0
noops=[]
with tempfile.TemporaryDirectory(prefix='cs467-independent-check-') as tmp:
 tmp=pathlib.Path(tmp)
 for p,b in RUNTIME.items():
  out=tmp/p;out.parent.mkdir(parents=True,exist_ok=True);out.write_bytes(b)
 for token,meta in MAN['cases'].items():
  cid=meta['case_id'];case=ROOT/'evals/golden/cases'/cid
  logs=case/'actual/assistant-dry-run/logs/run-1'
  ctx=json.loads((logs/'context.json').read_text())
  inp=json.loads((case/'input.json').read_text())
  inp_bytes=(case/'input.json').read_bytes()
  check(inp_bytes==git_bytes(f'evals/golden/cases/{cid}/input.json','4baa2cb'),f'frozen input: {cid}')
  check((case/'expected.json').read_bytes()==git_bytes(f'evals/golden/cases/{cid}/expected.json','4baa2cb'),f'frozen expected: {cid}')
  check(sha(inp_bytes)==meta['input_sha256'],f'input hash {cid}')
  issue_bytes=(logs/'issue.json').read_bytes();issue=json.loads(issue_bytes)
  expected_issue={'number':800000+int(token[1:]),'title':neutralise(inp['title']),'body':neutralise(inp['body']),'labels':inp['labels_at_open'],'state':'open','author_association':inp['author_association'],'as_of':inp['as_of'],'replay':True}
  check(issue==expected_issue,f'normalization {cid}')
  check(sha(issue_bytes)==meta['normalized_input_sha256']==ctx['file_sha256']['.triage/issue.json'],f'normalized hash {cid}')
  filtered=(''.join(l+'\n' for l in RUNTIME['data/issue-corpus.jsonl'].decode().splitlines() if l.strip() and json.loads(l)['created_at']<issue['as_of'])).encode()
  check(sha(filtered)==ctx['file_sha256']['data/issue-corpus.jsonl'],f'corpus hash {cid}')
  check(len(filtered.decode().splitlines())==ctx['earlier_corpus_rows'],f'corpus row count {cid}')
  expected_hashes={p:sha(b) for p,b in RUNTIME.items()}
  expected_hashes.update({'.triage/issue.json':sha(issue_bytes),'data/issue-corpus.jsonl':sha(filtered)})
  check(ctx['file_sha256']==expected_hashes,f'context hash inventory {cid}')
  check(ctx['source_commit']==SOURCE,f'context source {cid}')
  for p,expected_hash in ctx['file_sha256'].items():
   check(sha((TEMP/'cases'/token/'checkout'/p).read_bytes())==expected_hash,f'local prepared {token}/{p}')
  prepared_files={str(p.relative_to(TEMP/'cases'/token/'checkout')) for p in (TEMP/'cases'/token/'checkout').rglob('*') if p.is_file()}
  check(prepared_files==set(ctx['file_sha256']),f'extra files in prepared {cid}: {prepared_files-set(ctx["file_sha256"])}')
  for name in ['context.json','proposal.json','transcript.jsonl']:
   check((logs/name).read_bytes()==(TEMP/'cases'/token/name).read_bytes(),f'exported log {cid}/{name}')
  events=[json.loads(l) for l in (logs/'transcript.jsonl').read_text().splitlines()]
  check(sum(e['event']=='context' for e in events)==1 and events[0]['event']=='context',f'context first {cid}')
  check(sum(e['event']=='assistant_decision' for e in events)==1 and events[-1]['event']=='assistant_decision',f'decision final {cid}')
  times=[dt.datetime.fromisoformat(e['time_utc']) for e in events]
  check(times==sorted(times) and len(set(times))==len(times),f'chronology {cid}')
  check(times[-1]<dt.datetime.fromisoformat(SEAL['sealed_at_utc']),f'after seal {cid}')
  check(times[0]>BATCH_END.get(meta['batch'],dt.datetime.min.replace(tzinfo=dt.timezone.utc)),f'batch sequence {cid}')
  BATCH_END[meta['batch']]=times[-1]
  start_names=['AGENTS.md','CLAUDE.md','.github/workflows/issue-triage.md','.claude/skills/issue-triage/SKILL.md','.triage/issue.json']
  expected_context='\n\n'.join(f'--- {p} ---\n'+(issue_bytes if p=='.triage/issue.json' else RUNTIME[p]).decode() for p in start_names)
  check(events[0]['content']==expected_context,f'context displayed {cid}')
  for p,b in [('.triage/issue.json',issue_bytes),('data/issue-corpus.jsonl',filtered)]:
   out=tmp/p;out.parent.mkdir(parents=True,exist_ok=True);out.write_bytes(b)
  checks=[];valid_labels=set();valid_comments=[]
  for e in events[1:-1]:
   check(e['event']=='script',f'unexpected event {cid}')
   cmd=e['command'];check(cmd[0]=='python3' and cmd[1] in [f'.claude/skills/issue-triage/scripts/{s}.py' for s in ['triage_checks','check_labels','render_comment']],f'command {cid}')
   result=subprocess.run([sys.executable,*cmd[1:]],cwd=tmp,capture_output=True,text=True,env={'PATH':os.defpath,'HOME':str(tmp)},timeout=30)
   check((result.returncode,result.stdout,result.stderr)==(e['exit_code'],e['stdout'],e['stderr']),f'helper rerun mismatch {cid}: {cmd}')
   script_count+=1
   if cmd[1].endswith('/triage_checks.py'):
    check(cmd[2:]==[],f'input override {cid}')
    checks.append(json.loads(e['stdout'].split('\n',1)[1]))
   if cmd[1].endswith('/check_labels.py') and e['exit_code']==0: valid_labels.update(cmd[2:])
   if cmd[1].endswith('/render_comment.py') and e['exit_code']==0:valid_comments.append(e['stdout'])
  check(len(checks)==1 and events[1]['command'][1].endswith('/triage_checks.py'),f'checks ordering {cid}')
  proposal=json.loads((logs/'proposal.json').read_text());actual=json.loads((case/'actual/assistant-dry-run/run-1.json').read_text())
  check(events[-1]['content']==proposal,f'proposal not first decision {cid}')
  for name,key in [('proposal.json','proposal_sha256'),('transcript.jsonl','transcript_sha256')]:
   check(sha((logs/name).read_bytes())==SEAL['sessions'][token][key],f'seal {cid}/{name}')
  check(actual['proposal_sha256']==sha((logs/'proposal.json').read_bytes()),f'outcome hash {cid}')
  check(actual['source_commit']==SOURCE and actual['case_token']==token and not actual['writes_executed'],f'outcome provenance {cid}')
  labels=[];comment=None
  for item in proposal['items']:
   if item['type']=='add_labels':
    labels=item['labels'];check(set(labels)<=valid_labels,f'unvalidated labels {cid}')
   elif item['type']=='triage_comment':
    rendered=subprocess.run([sys.executable,'.claude/skills/issue-triage/scripts/render_comment.py','--kinds',item['kinds'],'--refs',item['refs']],cwd=tmp,capture_output=True,text=True)
    check(rendered.returncode==0,f'export invalid comment {cid}')
    comment=rendered.stdout.split('-----BEGIN COMMENT-----\n',1)[1].split('\n-----END COMMENT-----',1)[0]
    check(any(comment in c for c in valid_comments),f'comment not previously rendered {cid}')
   elif item['type']=='noop':noops.append(cid)
   else:raise AssertionError(f'unknown output {cid}: {item}')
  check(labels==actual['labels_added'] and comment==actual['comment'] and actual['other_writes']==[],f'export altered decision {cid}')
  if checks[0]['skip'] is not None:check(len(events)==3 and proposal['items'][0]['type']=='noop',f'failure to stop on skip {cid}')
  ALL_EVENTS.extend(events)
  SUMMARY.append({'token':token,'id':cid,'event_count':len(events),'skip':checks[0]['skip'],'items':proposal['items'],'start':events[0]['time_utc'],'end':events[-1]['time_utc']})
 # Copy only needed scoring artifacts and run the frozen scorer against copied first outcomes.
 scorepath=tmp/'evals/tools/score.py';scorepath.parent.mkdir(parents=True,exist_ok=True);scorepath.write_bytes(git_bytes('evals/tools/score.py'))
 for meta in MAN['cases'].values():
  cid=meta['case_id']
  for name in ['input.json','expected.json','actual/assistant-dry-run/run-1.json']:
   out=tmp/'evals/golden/cases'/cid/name;out.parent.mkdir(parents=True,exist_ok=True);out.write_bytes((ROOT/'evals/golden/cases'/cid/name).read_bytes())
 result=subprocess.run([sys.executable,str(scorepath),'--runs','assistant-dry-run','--k','1'],cwd=tmp,capture_output=True,text=True)
 check(result.returncode==0,'scorer failed '+result.stderr)
 check(json.loads((tmp/'evals/golden/results/score-assistant-dry-run.json').read_text())==json.loads((ROOT/'evals/golden/results/score-assistant-dry-run.json').read_text()),'score mismatch')
 print(result.stdout)
report={'status':'all independent checks passed','runtime_files':len(RUNTIME),'cases':len(SUMMARY),'helper_calls_reproduced':script_count,'transcript_events':len(ALL_EVENTS),'noops':noops,'first_event':min(e['time_utc'] for e in ALL_EVENTS),'last_event':max(e['time_utc'] for e in ALL_EVENTS),'sealed_at_utc':SEAL['sealed_at_utc'],'cases_detail':SUMMARY}
pathlib.Path('/tmp/cs467-independent-dry-run-check-results.json').write_text(json.dumps(report,indent=2))
print(json.dumps({k:v for k,v in report.items() if k!='cases_detail'},indent=2))
